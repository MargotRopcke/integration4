## our database setup
create table public.location_vibes (
  location_id uuid not null,
  vibe_category_id bigint not null,
  constraint location_vibes_pkey primary key (location_id, vibe_category_id),
  constraint location_vibes_location_id_fkey foreign KEY (location_id) references locations ("keyID") on delete CASCADE,
  constraint location_vibes_vibe_category_id_fkey foreign KEY (vibe_category_id) references vibe_categories (id) on delete CASCADE
) TABLESPACE pg_default;

create table public.location_vibes (
  location_id uuid not null,
  vibe_category_id bigint not null,
  constraint location_vibes_pkey primary key (location_id, vibe_category_id),
  constraint location_vibes_location_id_fkey foreign KEY (location_id) references locations ("keyID") on delete CASCADE,
  constraint location_vibes_vibe_category_id_fkey foreign KEY (vibe_category_id) references vibe_categories (id) on delete CASCADE
) TABLESPACE pg_default;

create table public.primary_categories (
  id bigint generated always as identity not null,
  name text not null,
  constraint primary_categories_pkey primary key (id),
  constraint primary_categories_name_key unique (name)
) TABLESPACE pg_default;

create table public.primary_categories (
  id bigint generated always as identity not null,
  name text not null,
  constraint primary_categories_pkey primary key (id),
  constraint primary_categories_name_key unique (name)
) TABLESPACE pg_default;

create table public.sessions_photos (
  id bigint generated by default as identity not null,
  user_id numeric not null,
  created_at timestamp with time zone not null default now(),
  photo text null,
  location_id uuid not null,
  constraint sessions_photos_pkey primary key (id),
  constraint sessions_photos_location_id_fkey foreign KEY (location_id) references locations ("keyID"),
  constraint sessions_photos_user_id_fkey foreign KEY (user_id) references sessions (user_id)
) TABLESPACE pg_default;

create table public.specialization_categories (
  id bigint generated by default as identity not null,
  name text null,
  image text null,
  constraint specialization_categories_pkey primary key (id)
) TABLESPACE pg_default;

create table public.traveler_types (
  id bigint generated by default as identity not null,
  name text null,
  description text null,
  constraint traveler_types_pkey primary key (id)
) TABLESPACE pg_default;

create table public.vibe_categories (
  id bigint generated always as identity not null,
  name text not null,
  primary_category_id bigint null,
  constraint vibe_categories_pkey primary key (id),
  constraint vibe_categories_name_key unique (name),
  constraint vibe_categories_primary_category_id_fkey foreign KEY (primary_category_id) references primary_categories (id)
) TABLESPACE pg_default;
